\author{Skip Potter}

\begin{abstract}
For primes $p\equiv1\pmod4$, define
\[
\sqdepth(p)=\min\{A\ge1:\exists D\mid A^2,\ 4A-1\mid p+4D\}.
\]
This is the first-hit depth in a classical divisor-square coordinate for prime Type-II Erd\H{o}s--Straus solutions. We study its exact-depth spectrum.

Every depth-$B$ trap residue $-4E$, $E\mid B^2$, has Jacobi sign $-1$ modulo $4B-1$. A parity-vector calculation shows that new global Jacobi-character directions first occur only at prime target moduli and at target moduli $3q$. Thus $3q$ is maximal for constructions that avoid all earlier layers solely by a global Jacobi-sign condition.

We introduce a finer fixed-cofactor mechanism. If a target residue is quadratic-residue at every fixed target prime and nonresidue at one variable prime $r$, all earlier layers not containing $r$ can be eliminated simultaneously. Every remaining $r$-bearing layer has modulus $cr$, and any collision forces
\[
r(kh-dc^2)+k+dh-2dc=0.
\]
A separate no-reuse theorem excludes $k=0$. Hence $k,h\ge1$ and AM--GM gives $kh<dc^2$, leaving only finitely many bad parameter primes.

This yields a canonical hierarchy for every fixed prime cofactor $s>3$. According to $s\pmod8$, an explicit witness $d_s$ and an explicit prime progression for $r$ give, apart from finitely many parameter primes, exact depths
\[
A=\frac{sr+1}{4}
\]
realized by infinitely many primes.

If $N_\infty(X)$ counts depth values $A\le X$ realized as exact square-completed Type-II depth by infinitely many primes, then
\[
\liminf_{X\to\infty}\frac{N_\infty(X)\log X}{X}\ge\frac83+C_{\rm all},
\]
where $C_{\rm all}>0$ is an explicitly convergent hierarchy series. Using only fixed prime cofactors $s\le10^6$ gives
\[
\boxed{\liminf_{X\to\infty}\frac{N_\infty(X)\log X}{X}\ge3.0353080580213\ldots .}
\]
These results concern exact Type-II depth and do not prove universal Type-II solvability or the Erd\H{o}s--Straus conjecture.
\end{abstract}

\section{Introduction}
The Erd\H{o}s--Straus conjecture asks whether
\[
\frac4n=\frac1x+\frac1y+\frac1z
\]
has a solution in positive integers for every $n\ge2$. We study prime Type-II solutions through the divisor-square coordinate
\begin{equation}\label{eq:square}
D\mid A^2,\qquad 4A-1\mid p+4D.
\end{equation}
The coordinate is classical and is not claimed here. Modern complete Type-II formulations appear in Elsholtz--Tao and Pomerance--Weingartner; Dahan studies Type-II sieve dimension and search depth.

For primes $p\equiv1\pmod4$, let $\sqdepth(p)$ be the least $A$ satisfying \eqref{eq:square}, and let
\[
N_\infty(X)=\#\{A\le X:\text{ infinitely many primes }p\text{ satisfy }\sqdepth(p)=A\}.
\]
Our problem is exact first occurrence rather than universal solvability.

\section{The complete square coordinate and Jacobi obstruction}
\begin{proposition}[complete prime Type-II coordinate]\label{prop:complete}
For a prime $p$, a Type-II solution exists if and only if there are positive integers $A,D$ with
\[
D\mid A^2,\qquad 4A-1\mid p+4D.
\]
For the reduced prime progressions used below one also has $p\nmid D$.
\end{proposition}

\begin{proof}
Pomerance--Weingartner, Corollary 2.4 specialized to $m=4$, states that a prime $p$ has a Type-II solution if and only if there are positive integers $a,b,e$ with
\[
e\mid a+b,\qquad 4ab\mid p+e.
\]
Assume these conditions and put
\[
c=\frac{a+b}{e},\qquad \alpha=\frac{p+e}{4ab},\qquad A=\alpha ac,\qquad D=\alpha a^2.
\]
Then $A^2/D=\alpha c^2$ is integral, so $D\mid A^2$. Since $b=ce-a$,
\[
\alpha ab+D=\alpha a(ce-a)+\alpha a^2=Ae,
\]
and therefore
\[
p+4D=4(\alpha ab+D)-e=(4A-1)e.
\]

Conversely, suppose $D\mid A^2$ and $p+4D=(4A-1)q$ for some $q>0$. Put $E=A^2/D$ and $\alpha=\gcd(D,E)$. Since $DE=A^2$ is a square and $D/\alpha,E/\alpha$ are coprime, there are coprime $u,c>0$ such that
\[
D=\alpha u^2,\qquad E=\alpha c^2,\qquad A=\alpha uc.
\]
Then
\[
p+q=4\alpha u(cq-u).
\]
Let $t=cq-u>0$. We have $4ut\mid p+q$ and $q\mid u+t$, so the Pomerance--Weingartner criterion holds with $(a,b,e)=(u,t,q)$.
\end{proof}

For $B\ge1$, put $m_B=4B-1$ and
\[
R_B=\{-4E\pmod{m_B}:E\mid B^2\}.
\]

\begin{lemma}[Jacobi obstruction]\label{lem:jacobi}
For every $E\mid B^2$,
\[
\left(\frac{-4E}{m_B}\right)=-1.
\]
\end{lemma}

\begin{proof}
Every prime divisor of $B$ is coprime to $m_B$. Quadratic reciprocity applied to $4B\equiv1\pmod{m_B}$ gives $(E/m_B)=1$ for every $E\mid B^2$, with the supplementary law handling the factor $2$. Since $m_B\equiv3\pmod4$, $(-1/m_B)=-1$.
\end{proof}

\section{The Jacobi-safe backbone}
For primes $q\equiv1\pmod4$, put $A_q=(3q+1)/4$.

\begin{theorem}[composite $3q$ backbone]\label{thm:3q}
If some $D\mid A_q^2$ satisfies $D\equiv2\pmod3$, then infinitely many primes $p$ have
\[
\sqdepth(p)=A_q.
\]
\end{theorem}

The proof uses CRT with residue $1$ on all earlier moduli; no earlier modulus can contain the factor $q$. A half-dimensional upper-bound sieve shows that the uncertified prime parameters have size $O(Y(\log Y)^{-3/2})$, so this composite backbone contributes $(2/3+o(1))X/\log X$ exact depths.

Write $v_A=(\operatorname{ord}_r(4A-1)\bmod2)_r$.

\begin{theorem}[first-occurrence Jacobi span]\label{thm:span}
The vector $v_A$ lies outside the span of $\{v_B:B<A\}$ if and only if either $4A-1$ is prime or
\[
4A-1=3q,\qquad q\equiv1\pmod4\text{ prime}.
\]
\end{theorem}

Thus $3q$ is maximal among composite targets for a construction using only Jacobi sign $+1$ on all earlier layers. The construction below goes beyond that barrier.

\section{One nonsquare coordinate and the master equation}
\begin{lemma}[single-nonsquare reduction]\label{lem:single}
Suppose the squarefree target modulus is $sr$, with $r$ prime, and the target residue $-4d$ is a quadratic residue at every prime divisor of the fixed squarefree cofactor $s$ but a nonresidue modulo $r$. Then every earlier layer whose modulus is not divisible by $r$ can be eliminated simultaneously by choosing square-safe CRT coordinates away from $r$.
\end{lemma}

\begin{proof}
Choose a unit square at every non-$r$ coordinate, preserving the target residue on common target factors. Every earlier modulus not containing $r$ then has Jacobi sign $+1$, and Lemma~\ref{lem:jacobi} excludes its trap set.
\end{proof}

Fix a squarefree odd cofactor $s$, a positive integer $d$ coprime to $s$, and a variable prime $r$. Put $A=(sr+1)/4$ and suppose $d\mid A$. An earlier modulus containing $r$ has the form
\[
4B-1=cr,\qquad 0<c<s.
\]

\begin{lemma}[master collision equation]\label{lem:master}
If the target branch collides with the depth-$B$ trap, then there are nonnegative integers $k,h$ satisfying
\[
r(kh-dc^2)+k+dh-2dc=0
\]
and
\[
16d\mid rh+1.
\]
If $k,h\ge1$ and the equality configuration is excluded, then $kh<dc^2$.
\end{lemma}

\begin{proof}
A collision gives $E\mid B^2$ with $E\equiv d\pmod r$, so $E=rk+d$. With $F=B^2/E$, reduction modulo $r$ gives $16dF=rh+1$. Multiplying
\[
(rk+d)(rh+1)=d(cr+1)^2
\]
yields the equation. If $kh>dc^2$, AM--GM gives $k+dh>2dc$, contradicting its sign. Equality forces $k=dc$, $h=c$, hence $16d\mid cr+1$.
\end{proof}

The condition $h\ge1$ is automatic. The remaining issue is $k=0$, meaning reuse of the target divisor $E=d$.

\section{The canonical no-reuse theorem}
For $d>0$, define
\[
\src(d)=\prod_p p^{\lceil v_p(d)/2\rceil}.
\]
Then $d\mid B^2$ if and only if $\src(d)\mid B$. Under the square-safe assignment $1$ at every non-target coordinate, $k=0$ can collide at $cr$ only if
\[
d\mid B^2\qquad\text{and}\qquad c\mid4d+1.
\]

$s\pmod8$ & witness $d_s$ & canonical prime progression for $r$ \\
\hline
$1$ & $2^{v_2(s-1)-3}(s-1)$ & $r\equiv-s^{-1}\pmod{4d_s}$ \\
$3$ & $(s+1)/2$ & $r\equiv s+2\pmod{2(s+1)}$ \\
$5$ & $s-1$ & $r\equiv s-2\pmod{4(s-1)}$ \\
$7$ & $(s-1)/2$ & $r\equiv s-2\pmod{2(s-1)}$
\end{tabular}
\caption{Canonical fixed-prime data.}
\label{tab:canonical}
\end{table}

\begin{theorem}[canonical no-reuse]\label{thm:noreuse}
For every fixed prime $s>3$, choose $d=d_s$ and the parameter progression in Table~\ref{tab:canonical}. Then no exceptional cofactor $c<s$ can satisfy both $d\mid B^2$ and $c\mid4d+1$.
\end{theorem}

\begin{proof}
We give the four congruence cases.

\emph{Case $s\equiv3\pmod8$.} Here $d=(s+1)/2=2m$ with $m$ odd and $r\equiv2d+1\pmod{4d}$. Write $m=u_0w_0$, where $u_0=\src(m)$; then $w_0\le u_0$. Because $s>3$, one has $m>1$, hence $u_0>1$. If $d\mid B^2$, then $B$ is even, hence $c\equiv3\pmod8$, and the odd part gives $u_0\mid c+1$. If also $4d+1=8m+1=ct$, then $t\equiv-1\pmod{u_0}$ and $t\equiv3\pmod8$. Thus
\[
c=u_0a-1,\qquad t=u_0b-1,\qquad a,b\ge4.
\]
Expanding gives
\[
ab u_0-(a+b)=8w_0.
\]
The left side is at least $16u_0-8>8u_0\ge8w_0$, contradiction.

\emph{Case $s\equiv7\pmod8$.} Here $d=(s-1)/2$ is odd and $r\equiv2d-1\pmod{4d}$. Put $u=\src(d)$ and $d=uw$, with $w\le u$. If $d\mid B^2$, then $u\mid B$, whence $u\mid c-1$. If $4d+1=ct$, then $t\equiv1\pmod u$; since $c,t\equiv3\pmod4$,
\[
c=ua+1,\qquad t=ub+1,\qquad a,b\ge2.
\]
Thus
\[
ab u+(a+b)=4w,
\]
whose left side is at least $4u+4$ while the right side is at most $4u$, contradiction.

\emph{Case $s\equiv5\pmod8$.} Take $d=s-1=4m$ with $m$ odd and $r\equiv d-1\pmod{4d}$. Write $m=u_0w_0$ with $u_0=\src(m)$ and $w_0\le u_0$. Exceptional cofactors satisfy $c\equiv1\pmod4$. The case $c=1$ cannot reuse $d$ because $B=(r+1)/4\equiv d/4\pmod d$ has $v_2(B)=0$ while $v_2(d)=2$. For $c>1$, $d\mid B^2$ implies $u_0\mid c-1$. If $4d+1=16m+1=ct$, then also $t\equiv1\pmod{u_0}$, and $c,t\equiv1\pmod4$. Hence
\[
c=u_0a+1,\qquad t=u_0b+1,\qquad a,b\ge4.
\]
Expansion gives
\[
ab u_0+(a+b)=16w_0,
\]
where the left side is at least $16u_0+8>16w_0$, contradiction.

\emph{Case $s\equiv1\pmod8$.} Let $e=v_2(s-1)\ge3$, write $s-1=2^e m$ with $m$ odd, and use
\[
d=2^{e-3}(s-1)=2^{2e-3}m.
\]
Write $m=u_0w_0$ with $u_0=\src(m)$ and $w_0\le u_0$; then $\src(d)=2^{e-1}u_0$. If $d\mid B^2$, then $2^{e+1}u_0\mid cr+1$. The parameter class gives $r\equiv-1\pmod{u_0}$ and $r\equiv2^e-1\pmod{2^{e+1}}$, so
\[
u_0\mid c-1,\qquad c\equiv2^e+1\pmod{2^{e+1}}.
\]
If also $4d+1=ct$, the same congruences imply
\[
t\equiv2^e+1\pmod{2^{e+1}}.
\]
Hence
\[
c=u_0a+1,\qquad t=u_0b+1,\qquad a,b\ge2^e.
\]
Since $4d+1=2^{2e-1}u_0w_0+1$,
\[
ab u_0+(a+b)=2^{2e-1}w_0.
\]
The left side is at least $2^{2e}u_0+2^{e+1}>2^{2e-1}u_0\ge2^{2e-1}w_0$, contradiction. This also excludes $c=1$.
\end{proof}

Consequently every actual collision has $k,h\ge1$, and after excluding the equality congruence it satisfies $kh<dc^2$. For each fixed $c$ there are only finitely many pairs $(k,h)$, and each determines at most one rational parameter
\[
r=\frac{k+dh-2dc}{dc^2-kh}.
\]

\section{A hierarchy for every prime fixed cofactor}
\begin{theorem}[every-prime fixed-cofactor hierarchy]\label{thm:allprime}
For every fixed prime $s>3$, use the canonical data in Table~\ref{tab:canonical}. Then all but finitely many prime parameters in the stated progression generate
\[
A=(sr+1)/4
\]
as an exact square-completed Type-II depth realized by infinitely many primes.
\end{theorem}

\begin{proof}
In each row, the parameter congruence gives $d_s\mid A$. Direct quadratic reciprocity shows that $-d_s$ is a residue modulo the fixed prime $s$ and a nonresidue modulo the variable prime $r$. The equality configuration $16d_s\mid cr+1$ is excluded by the same fixed congruence data. Lemma~\ref{lem:single} removes all non-$r$ layers, Theorem~\ref{thm:noreuse} removes $k=0$, and Lemma~\ref{lem:master} leaves only finitely many remaining parameter collisions. Deleting them leaves infinitely many parameter primes by Dirichlet. For each such parameter, the final reduced CRT class for the denominator is an arithmetic progression, so Dirichlet gives infinitely many exact-depth primes.
\end{proof}

\section{Counting the hierarchy}
Define
\[
c(s)=
\begin{cases}
\dfrac{4}{s\varphi(4\cdot2^{v_2(s-1)-3}(s-1))},&s\equiv1\pmod8,\\[6pt]
\dfrac{4}{s\varphi(2(s+1))},&s\equiv3\pmod8,\\[6pt]
\dfrac{4}{s\varphi(4(s-1))},&s\equiv5\pmod8,\\[6pt]
\dfrac{4}{s\varphi(2(s-1))},&s\equiv7\pmod8.
\end{cases}
\]
and
\[
C_{\rm all}=\sum_{\substack{s>3\\s\text{ prime}}}c(s).
\]
The series converges, since $c(s)\ll(\log\log s)/s^2$.

\begin{theorem}[spectrum lower bound]\label{thm:count}
\[
\liminf_{X\to\infty}\frac{N_\infty(X)\log X}{X}\ge\frac83+C_{\rm all}.
\]
Using only fixed prime cofactors $s\le10^6$ gives
\[
\boxed{\liminf_{X\to\infty}\frac{N_\infty(X)\log X}{X}\ge3.0353080580213976\ldots .}
\]
\end{theorem}

\begin{proof}
For any fixed finite set of cofactors, the prime number theorem in fixed arithmetic progressions applies separately to every family. Distinct fixed cofactors give distinct semiprime target moduli except for finitely many swapped-prime coincidences, which do not affect the asymptotic. Thus the contributions add for a fixed finite set. Taking $X\to\infty$ first and then the supremum over finite cofactor sets gives $C_{\rm all}$ without any growing-modulus uniformity.

The one-million finite certificate gives classwise partial sums
\[
0.006681657710714896,\quad
0.06768134901831344,\quad
0.12763386380907094,\quad
0.16664452081663197
\]
for $s\equiv1,3,5,7\pmod8$. Their sum is $0.36864139135473123\ldots$, which added to $8/3$ gives the displayed value.
\end{proof}

\section{Context and claim boundaries}
The accompanying repository contains independent exact certificates for composite layers such as $A=649$ and $A=22339$; these are not inputs to the hierarchy theorem. The $3q$ backbone also carries a separate half-dimensional sieve and Erd\H{o}s--Kac multiplicity theory.

The divisor-square coordinate, quadratic reciprocity, Dirichlet's theorem, prime number theorems in arithmetic progressions, and the analytic sieve machinery used for the $3q$ family are prior mathematics. Contemporary FCF/CENTL work contains closely related divisor-Jacobi, fixed-shift, ancestry, shadow, and exact-depth machinery. The candidate contribution here is the exact-first-hit fixed-cofactor mechanism: one nonsquare coordinate, the master collision equation, the no-reuse theorem, and the resulting hierarchy for every fixed prime cofactor. Targeted searches through August 2026 have not located this exact theorem; a negative search does not establish priority.

These results do not prove universal Type-II solvability and do not prove the Erd\H{o}s--Straus conjecture.
